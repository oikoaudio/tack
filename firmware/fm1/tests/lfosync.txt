# An LFO's sync is on or off, FREQ is a rate or a fraction, and each turn of SYNC takes the other's nearest speed
# firmware/fm1/build/host/tack_host firmware/fm1/tests/lfosync.txt OUTDIR
wait 600
tap LFO
wait 60

# FREQ 80 is 51 Hz. At 120 the nearest fraction is 1/32, 16 Hz, as the fastest there is
turn K3 80
wait 40
expect kitv115 80
expect kitv179 0
press OCT-
wait 30
turn K1 1
wait 40
release OCT-
wait 40
expect kitv179 11
expect kitv115 80
